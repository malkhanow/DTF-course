// ===================================================================
// Код для Google Apps Script — превращает Google Таблицу в мини-базу
// данных клиентов курса. Инструкция по установке — в конце файла.
// ===================================================================

const SECRET_TOKEN = 'ЗАМЕНИ_НА_СВОЙ_СЕКРЕТНЫЙ_КОД'; // придумай любой набор букв/цифр, никому его не давай

function doGet(e) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);

  if (body.token !== SECRET_TOKEN) {
    return ContentService.createTextOutput(JSON.stringify({ error: 'unauthorized' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const sheet = getSheet();
  sheet.appendRow([
    body.login,
    body.password,
    body.tier,
    body.access,
    body.name || '',
    body.telegram || '',
    body.email || '',
    new Date()
  ]);

  return ContentService.createTextOutput(JSON.stringify({ success: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName('Clients');
  if (!sheet) {
    sheet = ss.insertSheet('Clients');
    sheet.appendRow(['login', 'password', 'tier', 'access', 'name', 'telegram', 'email', 'date']);
  }
  return sheet;
}
